# 化學元素 ARPG「元素獵人」Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single-file HTML ARPG game where players explore randomly generated rooms, fight element-themed monsters, collect chemical element letters, and synthesize compounds for the codex and skills.

**Architecture:** Single HTML file (`ai_game/chemistry_arpg.html`) with inline CSS and JS. Canvas 2D for game rendering, DOM for HUD/menus. All game state managed in a single IIFE. localStorage for persistent save data.

**Tech Stack:** Vanilla HTML/CSS/JS, Canvas 2D API, Web Audio API, localStorage

## Global Constraints

- Single HTML file, all CSS/JS inlined — no external dependencies or libraries
- Canvas 2D rendering, no WebGL
- Dark theme matching existing games (background `#090d16`, card `#0f172a`, border `#334155`)
- Traditional Chinese (zh-Hant) for all UI text
- Mobile-first responsive: touch controls + keyboard/mouse
- Target 60fps, max 5 monsters per room
- Room grid: 16 columns × 12 rows, tile size auto-calculated from viewport
- Map: 5×5 rooms (25 total)
- localStorage key: `chem_arpg_save`

```js
// === MAP ===
let rooms = [];

function generateMap() {
  rooms = Array.from({ length: MAP_H }, () => Array.from({ length: MAP_W }, () => ({
    tiles: [], type: 'normal', generated: false, monsters: [], drops: [], cleared: false
  })));
  // Start room at center
  rooms[2][2].type = 'start';
  // Boss room on a random edge
  const edges = [];
  for (let y = 0; y < MAP_H; y++) for (let x = 0; x < MAP_W; x++) {
    if (x === 0 || x === MAP_W - 1 || y === 0 || y === MAP_H - 1) {
      if (!(x === 2 && y === 2)) edges.push([x, y]);
    }
  }
  const [bx, by] = edges[Math.floor(Math.random() * edges.length)];
  rooms[by][bx].type = 'boss';
  // 2-3 treasure rooms (not start, not boss)
  let treasureCount = 0;
  const target = 2 + Math.floor(Math.random() * 2);
  while (treasureCount < target) {
    const tx = Math.floor(Math.random() * MAP_W);
    const ty = Math.floor(Math.random() * MAP_H);
    if (rooms[ty][tx].type === 'normal') {
      rooms[ty][tx].type = 'treasure';
      treasureCount++;
    }
  }
}

function generateRoom(mx, my) {
  const room = rooms[my][mx];
  if (room.generated) return;
  room.generated = true;
  const tiles = Array.from({ length: ROWS }, () => Array(COLS).fill(TILE.FLOOR));
  // Border walls (leave exits open at center of each side)
  for (let c = 0; c < COLS; c++) { tiles[0][c] = TILE.WALL; tiles[ROWS - 1][c] = TILE.WALL; }
  for (let r = 0; r < ROWS; r++) { tiles[r][0] = TILE.WALL; tiles[r][COLS - 1] = TILE.WALL; }
  // Exits: 2-tile wide openings at center of each border if adjacent room exists
  if (my > 0) { tiles[0][7] = TILE.FLOOR; tiles[0][8] = TILE.FLOOR; }
  if (my < MAP_H - 1) { tiles[ROWS - 1][7] = TILE.FLOOR; tiles[ROWS - 1][8] = TILE.FLOOR; }
  if (mx > 0) { tiles[5][0] = TILE.FLOOR; tiles[6][0] = TILE.FLOOR; }
  if (mx < MAP_W - 1) { tiles[5][COLS - 1] = TILE.FLOOR; tiles[6][COLS - 1] = TILE.FLOOR; }
  // Random terrain for non-start rooms
  if (room.type !== 'start') {
    const obstacleCount = 8 + Math.floor(Math.random() * 12);
    for (let i = 0; i < obstacleCount; i++) {
      const r = 2 + Math.floor(Math.random() * (ROWS - 4));
      const c = 2 + Math.floor(Math.random() * (COLS - 4));
      if (tiles[r][c] === TILE.FLOOR) {
        const types = [TILE.ROCK, TILE.ROCK, TILE.GRASS, TILE.GRASS, TILE.WATER, TILE.WALL];
        tiles[r][c] = types[Math.floor(Math.random() * types.length)];
      }
    }
  }
  room.tiles = tiles;
}

function drawRoom() {
  const room = rooms[state.mapY][state.mapX];
  if (!room.generated) return;
  for (let r = 0; r < ROWS; r++) {
    for (let c = 0; c < COLS; c++) {
      const t = room.tiles[r][c];
      ctx.fillStyle = TILE_COLORS[t];
      ctx.fillRect(c * tileSize, r * tileSize, tileSize, tileSize);
      // Grid line
      ctx.strokeStyle = 'rgba(30,41,59,0.4)';
      ctx.lineWidth = 0.5;
      ctx.strokeRect(c * tileSize, r * tileSize, tileSize, tileSize);
      // Rock/grass decorations
      if (t === TILE.ROCK) {
        ctx.fillStyle = '#64748b';
        ctx.font = `${tileSize * 0.5}px sans-serif`;
        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
        ctx.fillText('🪨', c * tileSize + tileSize / 2, r * tileSize + tileSize / 2);
      } else if (t === TILE.GRASS) {
        ctx.fillStyle = '#22c55e';
        ctx.font = `${tileSize * 0.45}px sans-serif`;
        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
        ctx.fillText('~', c * tileSize + tileSize / 2, r * tileSize + tileSize / 2);
      } else if (t === TILE.WATER) {
        ctx.fillStyle = '#38bdf8';
        ctx.font = `${tileSize * 0.45}px sans-serif`;
        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
        ctx.fillText('≋', c * tileSize + tileSize / 2, r * tileSize + tileSize / 2);
      }
    }
  }
}

function transitionRoom(dx, dy) {
  const nx = state.mapX + dx;
  const ny = state.mapY + dy;
  if (nx < 0 || nx >= MAP_W || ny < 0 || ny >= MAP_H) return false;
  state.mapX = nx;
  state.mapY = ny;
  generateRoom(nx, ny);
  return true;
}
```

```js
function initPlayer() {
  player.x = 7.5; player.y = 6;
  player.dir = 'down';
  player.hp = player.maxHp;
  player.mp = player.maxMp;

function isSolid(col, row) {
  if (col < 0 || col >= COLS || row < 0 || row >= ROWS) return true;
  const t = rooms[state.mapY][state.mapX].tiles[row][col];
  return t === TILE.WALL || t === TILE.ROCK || t === TILE.WATER;
}

function updatePlayer(dt) {
  let dx = 0, dy = 0;
  if (keys.up) { dy = -1; player.dir = 'up'; }
  if (keys.down) { dy = 1; player.dir = 'down'; }
  if (keys.left) { dx = -1; player.dir = 'left'; }
  if (keys.right) { dx = 1; player.dir = 'right'; }
  // Normalize diagonal
  if (dx && dy) { dx *= 0.707; dy *= 0.707; }
  const spd = player.speed * dt;
  const nx = player.x + dx * spd;
  const ny = player.y + dy * spd;
  const halfW = 0.35;
  // X collision
  if (dx !== 0) {
    const testX = dx > 0 ? nx + halfW : nx - halfW;
    if (!isSolid(Math.floor(testX), Math.floor(player.y - halfW)) &&
        !isSolid(Math.floor(testX), Math.floor(player.y + halfW - 0.01))) {
      player.x = nx;
    }
  }
  // Y collision
  if (dy !== 0) {
    const testY = dy > 0 ? ny + halfW : ny - halfW;
    if (!isSolid(Math.floor(player.x - halfW), Math.floor(testY)) &&
        !isSolid(Math.floor(player.x + halfW - 0.01), Math.floor(testY))) {
      player.y = ny;
    }
  }
  // Room transition
  if (player.x < 0.1) { if (transitionRoom(-1, 0)) player.x = COLS - 1.5; }
  else if (player.x > COLS - 0.1) { if (transitionRoom(1, 0)) player.x = 1.5; }
  else if (player.y < 0.1) { if (transitionRoom(0, -1)) player.y = ROWS - 1.5; }
  else if (player.y > ROWS - 0.1) { if (transitionRoom(0, 1)) player.y = 1.5; }
  // Timers
  if (player.atkTimer > 0) player.atkTimer -= dt;
  if (player.invincible > 0) player.invincible -= dt;
  if (player.shieldTimer > 0) player.shieldTimer -= dt;
}
```

```js
// === MONSTERS ===
function spawnMonsters(room, mx, my) {
  if (room.type === 'start') return;
  room.monsters = [];
  const count = room.type === 'boss' ? 1 : (room.type === 'treasure' ? 3 : 2 + Math.floor(Math.random() * 4));
  for (let i = 0; i < count; i++) {
    // Find open floor tile
    let col, row;
    do {
      col = 2 + Math.floor(Math.random() * (COLS - 4));
      row = 2 + Math.floor(Math.random() * (ROWS - 4));
    } while (room.tiles[row][col] !== TILE.FLOOR);
    const isBoss = room.type === 'boss' && i === 0;
    const isMolecular = !isBoss && Math.random() < 0.15 && i === 0;
    const el = ELEMENTS[Math.floor(Math.random() * ELEMENTS.length)];
    const lvScale = 1 + (player.level - 1) * 0.15;
    if (isBoss) {
      room.monsters.push({ x: col + 0.5, y: row + 0.5, hp: Math.floor(80 * lvScale), maxHp: Math.floor(80 * lvScale),
        atk: Math.floor(12 * lvScale), speed: 1.5, element: el, type: 'boss',
        dotTimer: 0, frozen: 0, shootTimer: 2, dir: 'down' });
    } else if (isMolecular) {
      const recipe = RECIPES[Math.floor(Math.random() * RECIPES.length)];
      room.monsters.push({ x: col + 0.5, y: row + 0.5, hp: Math.floor(30 * lvScale), maxHp: Math.floor(30 * lvScale),
        atk: Math.floor(6 * lvScale), speed: 1.2, element: el, type: 'molecular', recipe,
        dotTimer: 0, frozen: 0 });
    } else {
      room.monsters.push({ x: col + 0.5, y: row + 0.5, hp: Math.floor(10 * lvScale), maxHp: Math.floor(10 * lvScale),
        atk: Math.floor(3 * lvScale), speed: 1.8, element: el, type: 'slime',
        dotTimer: 0, frozen: 0 });
    }
  }
}

function updateMonsters(dt) {
  const room = rooms[state.mapY][state.mapX];
  for (let i = room.monsters.length - 1; i >= 0; i--) {
    const m = room.monsters[i];
    // DoT
    if (m.dotTimer > 0) {
      m.dotTimer -= dt;
      m.hp -= player.atk * 0.3 * dt;
      if (m.hp <= 0) { killMonster(room, i); continue; }
    }
    // Frozen
    if (m.frozen > 0) { m.frozen -= dt; continue; }
    // Chase player
    const dx = player.x - m.x;
    const dy = player.y - m.y;
    const dist = Math.sqrt(dx * dx + dy * dy);
    if (dist < 8 && dist > 0.6) {
      const mx = (dx / dist) * m.speed * dt;
      const my2 = (dy / dist) * m.speed * dt;
      const nnx = m.x + mx;
      const nny = m.y + my2;
      if (!isSolid(Math.floor(nnx), Math.floor(m.y))) m.x = nnx;
      if (!isSolid(Math.floor(m.x), Math.floor(nny))) m.y = nny;
    }
    // Contact damage
    if (dist < 0.7 && player.invincible <= 0) {
      damagePlayer(m.atk);
    }
    // Boss shoots
    if (m.type === 'boss') {
      m.shootTimer = (m.shootTimer || 0) - dt;
      if (m.shootTimer <= 0 && dist < 10) {
        m.shootTimer = 2.5;
        const angle = Math.atan2(dy, dx);
        // Boss projectile reuses the projectiles array
        projectiles.push({ x: m.x, y: m.y, vx: Math.cos(angle + Math.PI) * 4,
          vy: Math.sin(angle + Math.PI) * 4, dmg: m.atk, fromPlayer: false, element: m.element, life: 3, pierce: false });
      }
    }
  }
}

function drawMonsters() {
  const room = rooms[state.mapY][state.mapX];
  room.monsters.forEach(m => {
    const px = m.x * tileSize;
    const py = m.y * tileSize;
    const radius = m.type === 'boss' ? tileSize * 0.65 : (m.type === 'molecular' ? tileSize * 0.5 : tileSize * 0.4);
    // Shadow
    ctx.fillStyle = 'rgba(0,0,0,0.3)';
    ctx.beginPath(); ctx.ellipse(px, py + radius * 0.6, radius * 0.7, radius * 0.2, 0, 0, Math.PI * 2); ctx.fill();
    // Body
    ctx.fillStyle = m.frozen > 0 ? '#93c5fd' : m.element.color;
    if (m.type === 'boss') {
      ctx.fillRect(px - radius, py - radius, radius * 2, radius * 2);
      // Flash border
      ctx.strokeStyle = Math.floor(Date.now() / 200) % 2 ? '#fbbf24' : '#ef4444';
      ctx.lineWidth = 3;
      ctx.strokeRect(px - radius, py - radius, radius * 2, radius * 2);
    } else {
      ctx.beginPath(); ctx.arc(px, py, radius, 0, Math.PI * 2); ctx.fill();
      ctx.strokeStyle = 'rgba(255,255,255,0.3)';
      ctx.lineWidth = 1.5;
      ctx.stroke();
    }
    // Symbol
    ctx.fillStyle = '#fff';
    ctx.font = `bold ${radius * 0.9}px sans-serif`;
    ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    ctx.fillText(m.element.sym, px, py);
    // HP bar
    if (m.hp < m.maxHp) {
      const barW = radius * 2;
      const barH = 4;
      const barX = px - radius;
      const barY = py - radius - 8;
      ctx.fillStyle = '#1e293b'; ctx.fillRect(barX, barY, barW, barH);
      ctx.fillStyle = '#ef4444'; ctx.fillRect(barX, barY, barW * (m.hp / m.maxHp), barH);
    }
  });
}

function damagePlayer(amount) {
  if (player.shieldTimer > 0 || player.invincible > 0) return;
  player.hp -= amount;
  player.invincible = 0.5;
  playTone(180, 'sawtooth', 0.12);
  if (player.hp <= 0) {
    player.hp = 0;
    playerDeath();
  }
}

function killMonster(room, index) {
  const m = room.monsters[index];
  // Spawn drops
  if (m.type === 'molecular' && m.recipe) {
    Object.entries(m.recipe.req).forEach(([sym]) => {
      room.drops.push({ x: m.x + (Math.random() - 0.5) * 0.8, y: m.y + (Math.random() - 0.5) * 0.8, sym, life: 15 });
    });
  } else if (m.type === 'boss') {
    const count = 3 + Math.floor(Math.random() * 3);
    for (let j = 0; j < count; j++) {
      const el = ELEMENTS[Math.floor(Math.random() * ELEMENTS.length)];
      room.drops.push({ x: m.x + (Math.random() - 0.5), y: m.y + (Math.random() - 0.5), sym: el.sym, life: 30 });
    }
  } else {
    room.drops.push({ x: m.x, y: m.y, sym: m.element.sym, life: 15 });
  }
  // EXP
  const expGain = m.type === 'boss' ? 100 : (m.type === 'molecular' ? 30 : 10 + Math.floor(Math.random() * 11));
  gainExp(expGain);
  room.monsters.splice(index, 1);
  playTone(520, 'sine', 0.08);
  saveGame();
}

// === MELEE ATTACK ===
function meleeAttack() {
  if (player.atkTimer > 0) return;
  player.atkTimer = 0.3;
  playTone(300, 'square', 0.06);
  const room = rooms[state.mapY][state.mapX];
  const angles = { up: -Math.PI / 2, down: Math.PI / 2, left: Math.PI, right: 0 };
  const atkAngle = angles[player.dir];
  for (let i = room.monsters.length - 1; i >= 0; i--) {
    const m = room.monsters[i];
    const dx = m.x - player.x;
    const dy = m.y - player.y;
    const dist = Math.sqrt(dx * dx + dy * dy);
    if (dist > 1.8) continue;
    const angle = Math.atan2(dy, dx);
    let diff = angle - atkAngle;
    while (diff > Math.PI) diff -= Math.PI * 2;
    while (diff < -Math.PI) diff += Math.PI * 2;
    if (Math.abs(diff) < 0.8) {
      m.hp -= player.atk;
      if (m.hp <= 0) killMonster(room, i);
      else playTone(400, 'triangle', 0.04);
    }
  }
  // Break rocks
  const dxDir = { up: 0, down: 0, left: -1, right: 1 };
  const dyDir = { up: -1, down: 1, left: 0, right: 0 };
  const rc = Math.floor(player.x + dxDir[player.dir]);
  const rr = Math.floor(player.y + dyDir[player.dir]);
  if (rc >= 0 && rc < COLS && rr >= 0 && rr < ROWS && room.tiles[rr][rc] === TILE.ROCK) {
    room.tiles[rr][rc] = TILE.FLOOR;
    if (Math.random() < 0.3) {
      const el = ELEMENTS[Math.floor(Math.random() * ELEMENTS.length)];
      room.drops.push({ x: rc + 0.5, y: rr + 0.5, sym: el.sym, life: 15 });
    }
  }
}
```

```js
let inventory = { H: 0, O: 0, C: 0, Na: 0, Cl: 0, Ca: 0 };

function cycleAmmo() {
  const syms = ELEMENTS.map(e => e.sym);
  let idx = syms.indexOf(selectedAmmo);
  for (let i = 1; i <= syms.length; i++) {

function updateDrops(dt) {

    const d = room.drops[i];
    d.life -= dt;
    if (d.life <= 0) { room.drops.splice(i, 1); continue; }
    const dx = player.x - d.x;

// === PROJECTILES ===
let projectiles = [];

function shootProjectile() {
  if (inventory[selectedAmmo] <= 0) return;
  inventory[selectedAmmo]--;
  updateElemBar();
  const el = ELEMENTS.find(e => e.sym === selectedAmmo);
  const dirs = { up: [0, -1], down: [0, 1], left: [-1, 0], right: [1, 0] };
  const [vx, vy] = dirs[player.dir];
  projectiles.push({
    x: player.x, y: player.y, vx: vx * el.speed, vy: vy * el.speed,
    dmg: player.atk * el.dmgMul, fromPlayer: true, element: el, life: 2.5,
    pierce: el.fx === 'pierce'
  });
  playTone(500, 'sine', 0.05);
}

function updateProjectiles(dt) {
  const room = rooms[state.mapY][state.mapX];
  for (let i = projectiles.length - 1; i >= 0; i--) {
    const p = projectiles[i];
    p.x += p.vx * dt;
    p.y += p.vy * dt;
    p.life -= dt;
    // Wall collision
    if (p.life <= 0 || isSolid(Math.floor(p.x), Math.floor(p.y))) {
      projectiles.splice(i, 1); continue;
    }
    if (p.fromPlayer) {
      // Hit monsters
      for (let j = room.monsters.length - 1; j >= 0; j--) {
        const m = room.monsters[j];
        const dx = m.x - p.x, dy = m.y - p.y;
        if (Math.sqrt(dx * dx + dy * dy) < 0.6) {
          m.hp -= p.dmg;
          // Special effects
          if (p.element.fx === 'knockback') {
            const kd = Math.sqrt(dx * dx + dy * dy) || 1;
            m.x += (dx / kd) * 2; m.y += (dy / kd) * 2;
          }
          if (p.element.fx === 'dot') m.dotTimer = 3;
          if (m.hp <= 0) killMonster(room, j);
          else playTone(400, 'triangle', 0.04);
          if (!p.pierce) { projectiles.splice(i, 1); break; }
        }
      }
    } else {
      // Enemy projectile hits player
      const dx = player.x - p.x, dy = player.y - p.y;
      if (Math.sqrt(dx * dx + dy * dy) < 0.6) {
        damagePlayer(p.dmg);
        projectiles.splice(i, 1);
      }
    }
  }
}

function drawProjectiles() {
  projectiles.forEach(p => {
    const px = p.x * tileSize, py = p.y * tileSize;

    ctx.textAlign = 'center'; ctx.textBaseline = 'middle';

  });
}

    el.addEventListener('click', () => { selectedAmmo = el.dataset.sym; updateElemBar(); });
  });
}
```

```js
let codex = [];

let equippedSkills = [null, null, null, null];

function synthesize(recipeId) {
  const recipe = RECIPES.find(r => r.id === recipeId);
  if (!recipe) return;
  // Check inventory
  for (const [sym, count] of Object.entries(recipe.req)) {
    if ((inventory[sym] || 0) < count) return;
  }
  // Consume
  for (const [sym, count] of Object.entries(recipe.req)) inventory[sym] -= count;
  // Unlock
  const firstTime = !codex.includes(recipeId);
  if (firstTime) {
    codex.push(recipeId);
    unlockedSkills.push(recipeId);
    // Auto-equip to first empty slot
    const emptySlot = equippedSkills.indexOf(null);
    if (emptySlot >= 0) equippedSkills[emptySlot] = recipeId;
    gainExp(recipe.pts);
    showToast(`圖鑑解鎖：${recipe.formula} ${recipe.name}！`);
  } else {
    gainExp(50);
    showToast(`合成 ${recipe.formula} +50 EXP`);

}
```

```js
  const skillId = equippedSkills[slotIdx];
  if (!skillId) return;
  const recipe = RECIPES.find(r => r.id === skillId);
  if (!recipe || player.mp < recipe.mp) return;

        if (dist < 3) m.hp -= dmg;

function updateTraps(dt) {
  const room = rooms[state.mapY][state.mapX];
  for (let i = traps.length - 1; i >= 0; i--) {
    traps[i].life -= dt;
    if (traps[i].life <= 0) { traps.splice(i, 1); continue; }
    room.monsters.forEach(m => {
      const dist = Math.sqrt((m.x - traps[i].x) ** 2 + (m.y - traps[i].y) ** 2);
      if (dist < 0.8) m.hp -= traps[i].dmg * dt;
    });
    for (let j = room.monsters.length - 1; j >= 0; j--) {
      if (room.monsters[j].hp <= 0) killMonster(room, j);
    }
  }
  // Acid zones
  for (let i = acidZones.length - 1; i >= 0; i--) {
    acidZones[i].life -= dt;
    if (acidZones[i].life <= 0) { acidZones.splice(i, 1); continue; }
    const az = acidZones[i];
    const dirs = { up: [0, -1], down: [0, 1], left: [-1, 0], right: [1, 0] };
    const [adx, ady] = dirs[az.dir];
    room.monsters.forEach(m => {
      const relX = m.x - az.x, relY = m.y - az.y;
      const proj = relX * adx + relY * ady;
      if (proj > 0 && proj < 3) {
        const perp = Math.abs(relX * ady - relY * adx);
        if (perp < 1.5) m.hp -= az.dmg * dt;
      }
    });
    for (let j = room.monsters.length - 1; j >= 0; j--) {
      if (room.monsters[j].hp <= 0) killMonster(room, j);
    }
  }
  // Smoke timer
  if (smokeTimer > 0) smokeTimer -= dt;
}

    ctx.fillText('☠', t.x * tileSize, t.y * tileSize);
  });
  acidZones.forEach(az => {
    const dirs = { up: [0, -1], down: [0, 1], left: [-1, 0], right: [1, 0] };
    const [adx, ady] = dirs[az.dir];

    ctx.lineTo((az.x + adx * 3 + ady * 1.5) * tileSize, (az.y + ady * 3 + adx * 1.5) * tileSize);

    ctx.closePath(); ctx.fill();
  });
```

```js
const SAVE_KEY = 'chem_arpg_save';

function saveGame() {
  const data = {
    player: { level: player.level, exp: player.exp, maxHp: player.maxHp, maxMp: player.maxMp,
      atk: player.atk, currentHp: player.hp, currentMp: player.mp },
    inventory: { ...inventory },
    codex: [...codex],
    unlockedSkills: [...unlockedSkills],
    equippedSkills: [...equippedSkills],
    selectedAmmo,
    stats: { totalKills: window._stats?.totalKills || 0, totalSyntheses: codex.length, deaths: window._stats?.deaths || 0 }
  };
  localStorage.setItem(SAVE_KEY, JSON.stringify(data));
}

function loadGame() {
  const raw = localStorage.getItem(SAVE_KEY);
  if (!raw) return;
  try {
    const data = JSON.parse(raw);
    if (data.player) {
      player.level = data.player.level || 1;
      player.exp = data.player.exp || 0;
      player.maxHp = data.player.maxHp || 50;
      player.maxMp = data.player.maxMp || 20;
      player.atk = data.player.atk || 5;
      player.hp = data.player.currentHp || player.maxHp;
      player.mp = data.player.currentMp || player.maxMp;
    }
    if (data.inventory) Object.assign(inventory, data.inventory);
    if (data.codex) codex = data.codex;
    if (data.unlockedSkills) unlockedSkills = data.unlockedSkills;
    if (data.equippedSkills) equippedSkills = data.equippedSkills;
    if (data.selectedAmmo) selectedAmmo = data.selectedAmmo;
    if (data.stats) window._stats = data.stats;
  } catch (e) { /* corrupted save, ignore */ }
}

const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
function playTone(freq, type = 'sine', dur = 0.08) {
  if (audioCtx.state === 'suspended') audioCtx.resume();
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type; osc.frequency.value = freq;
  gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
  osc.connect(gain); gain.connect(audioCtx.destination);
  osc.start(); osc.stop(audioCtx.currentTime + dur);
}
```
